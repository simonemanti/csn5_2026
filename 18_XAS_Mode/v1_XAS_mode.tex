% Beam dimensions: change these values to adjust the black outline and widths.
\newcommand{\beamOutline}{0.5mm}

\newcommand{\outlinedbeam}[4]{%
  \begin{scope}
    \coordinate (beamStart) at #3;
    \coordinate (beamEnd) at #4;
    \path let \p1 = ($(beamEnd)-(beamStart)$),
      \n1 = {atan2(\y1,\x1)}, \n2 = {veclen(\x1,\y1)} in
      \pgfextra{\xdef\beamAngle{\n1}\xdef\beamLength{\n2}};
    \pgfmathsetlengthmacro{\beamHalfWidth}{#1/2}
    \begin{scope}[shift={(beamStart)},rotate=\beamAngle]
      \path[fill=#2,draw=black,line width=\beamOutline,line join=round]
        (0,-\beamHalfWidth) -- (\beamLength-5mm,-\beamHalfWidth)
        -- (\beamLength-5mm,-4mm) -- (\beamLength,0)
        -- (\beamLength-5mm,4mm) -- (\beamLength-5mm,\beamHalfWidth)
        -- (0,\beamHalfWidth) -- cycle;
    \end{scope}
  \end{scope}
}
